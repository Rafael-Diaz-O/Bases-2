1. Películas después del año 2000 con rating alto

MATCH (u:Person)-[r:REVIEWED]->(m:Movie)
WHERE m.released > 2000 AND r.rating >= 8
RETURN 
    m.title AS Película,
    m.released AS Año,
    round(avg(r.rating), 2) AS PromedioRating,
    count(r) AS CantidadReseñas
ORDER BY PromedioRating DESC
LIMIT 10;

2. Actores cuyo nombre empieza por “M” y con más de 3
películas

MATCH (a:Person)-[:ACTED_IN]->(m:Movie) 
WHERE a.name STARTS WITH "M" 
WITH a, COUNT(m) AS totalPeliculas 
WHERE totalPeliculas > 3 
RETURN a.name, totalPeliculas 
ORDER BY totalPeliculas DESC;


3. Películas sin tagline

MATCH (m:Movie)
WHERE m.tagline IS NULL
RETURN m.title, m.released;

4.Caminos de actores conectados a “Tom Hanks” hasta 3
niveles


MATCH path = (a:Person {name: "Tom Hanks"})-[:ACTED_IN*1..3]-(b:Person)
RETURN DISTINCT b.name AS actoresRelacionados, length(path) AS niveles
ORDER BY niveles;

5. Actor y director en una misma película

MATCH (p:Person)-[:ACTED_IN]->(m:Movie)<-[:DIRECTED]-(p)
RETURN DISTINCT p.name AS ActorDirector, COLLECT(m.title) AS peliculas;

6.Clasificación de actores según cantidad de películas
(CASE)

MATCH (a:Person)-[:ACTED_IN]->(m:Movie)
WITH a, COUNT(m) AS totalPeliculas
RETURN a.name,
  CASE 
    WHEN totalPeliculas > 10 THEN "Veterano"
    WHEN totalPeliculas >= 5 THEN "Experimentado"
    ELSE "Nuevo"
  END AS categoria;




7. Marcar películas antiguas como OLD_MOVIE (FOREACH)

MATCH (m:Movie)
FOREACH (x IN CASE WHEN m.released < 2000 THEN [m] ELSE [] END |
  SET m:OLD_MOVIE);



8. Total de actores por director (agregación)

MATCH (d:Person)-[:DIRECTED]->(m:Movie)<-[:ACTED_IN]-(a:Person)
RETURN d.name AS Director, COUNT(DISTINCT a) AS ActoresTrabajaron
ORDER BY ActoresTrabajaron DESC;


9. Top 5 actores con más películas

MATCH (a:Person)-[:ACTED_IN]->(m:Movie)
RETURN a.name, COUNT(m) AS totalPeliculas
ORDER BY totalPeliculas DESC
LIMIT 5;


10. Subconsulta: total de películas actuadas por actor

MATCH (a:Person)
CALL {
  WITH a
  MATCH (a)-[:ACTED_IN]->(m:Movie)
  RETURN COUNT(m) AS total
}
RETURN a.name, total
ORDER BY total DESC
LIMIT 10;

11. Películas dirigidas y actuadas por la misma persona

MATCH (p:Person)-[:DIRECTED]->(m:Movie)<-[:ACTED_IN]-(p)
RETURN p.name AS Persona, COLLECT(m.title) AS Peliculas;


12. Actores conectados a través de caminos de longitud
variable (*..4)

MATCH path = (a:Person {name: "Keanu Reeves"})-[:ACTED_IN*..4]-(b:Person)
RETURN DISTINCT b.name AS ActoresRelacionados, length(path) AS profundidad
ORDER BY profundidad;
